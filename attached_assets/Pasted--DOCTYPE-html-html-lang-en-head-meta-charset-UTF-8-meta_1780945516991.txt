  // ---- stat point allocation (staged, then confirmed) ----------------
  function totalPending(){ return playerData.stats.reduce((a,s) => a + s.pending, 0); }

  // delta +1 stages a point (if pool has room); -1 removes a staged point only
  function adjust(i, delta){
    const d = playerData;
    if (delta > 0) {
      if (totalPending() >= d.availablePoints) return;   // whole pool already staged
      d.stats[i].pending++;
    } else {
      if (d.stats[i].pending <= 0) return;               // can't drop below committed base
      d.stats[i].pending--;
    }
    refreshStats();
  }

  // discard the unconfirmed batch (does NOT touch committed stats)
  function clearPending(){
    playerData.stats.forEach(s => s.pending = 0);
    refreshStats();
  }

  // lock the batch in. In production this is where the SERVER takes over:
  //   POST /api/player/:id/allocate  with the final per-stat deltas.
  //   The server validates the player truly has these points, applies them,
  //   recomputes combat power, and returns authoritative state to re-render.
  function commitPoints(){
    const d = playerData, spent = totalPending();
    if (spent <= 0) return;
    d.stats.forEach(s => { s.val += s.pending; s.pending = 0; });
    d.availablePoints -= spent;
    d.combatPower += spent * d.cpPerPoint;               // local feedback only
    document.getElementById('cp').textContent = d.combatPower.toLocaleString();
    refreshStats();
  }

  // single source of truth for the stats UI
  function refreshStats(){
    const d = playerData;
    const pending = totalPending();
    const remaining = d.availablePoints - pending;

    document.getElementById('availPts').textContent = remaining;
    document.getElementById('pointsbar').classList.toggle('empty', remaining <= 0);

    document.getElementById('statlist').innerHTML = d.stats.map((s, i) => {
      const valHtml = s.pending > 0
        ? `<span class="base">${s.val}</span><span class="arrow">→</span><span class="staged">${s.val + s.pending}</span>`
        : `${s.val}`;
      return `<div class="statcard">
        <div class="row">
          <svg viewBox="0 0 24 24" fill="none" stroke="${s.color}" stroke-width="2">${ICONS[s.icon]}</svg>
          <span class="nm" style="color:${s.color}">${s.key}</span>
          <span class="num">${valHtml}</span>
          <button class="minusbtn ${s.pending <= 0 ? 'disabled' : ''}" onclick="adjust(${i},-1)">−</button>
          <button class="plusbtn ${remaining <= 0 ? 'disabled' : ''}" onclick="adjust(${i},1)">+</button>
        </div>
      </div>`;
    }).join('');

    document.getElementById('confirmBtn').classList.toggle('disabled', pending <= 0);
    document.getElementById('cancelBtn').classList.toggle('disabled', pending <= 0);
  }
